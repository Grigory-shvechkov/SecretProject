# opencv-python is deliberately NOT listed here for pip to install.
# On this Pi (Python 3.9, armv7l), pip's source build fails patching
# OpenEXR -- there's no matching prebuilt wheel for this platform/Python
# combo. Install it via apt instead, which ships a precompiled binary:
#     sudo apt install -y python3-opencv
#
# numpy's piwheels wheel is built against OpenBLAS and expects
# libopenblas.so.0 to already exist on the system -- pip installing numpy
# succeeds, but `import numpy` (or anything that imports it, e.g. cv2)
# then fails with "ImportError: libopenblas.so.0: cannot open shared
# object file". Also an apt package, also install it before relying on
# this file:
#     sudo apt install -y libopenblas0
#
# Pinned to <2 (numpy==1.26.4, the last 1.x release, confirmed to have a
# cp39-armv7l wheel on piwheels) because NumPy 2.x broke C-API/ABI
# compatibility with anything compiled against NumPy 1.x -- including
# this Pi's apt-installed python3-opencv, which was built against a 1.x
# numpy. Installing numpy 2.x (what pip picks by default, unpinned) makes
# `import cv2` fail with "AttributeError: _ARRAY_API not found" even
# though numpy itself imports fine on its own. If python3-opencv is ever
# rebuilt/upgraded against NumPy 2.x, this pin can be dropped.
numpy==1.26.4
requests

# Fish detection runs on TensorFlow Lite, not PyTorch/ultralytics.
# PyTorch was tried first and is a dead end on this hardware: official
# PyTorch has never published wheels for 32-bit ARM (armv7l), at any
# version, on PyPI or piwheels.
#
# Pinned to 2.13.0 specifically, not the latest (2.14.0) -- confirmed by
# checking PyPI's file listing per-release:
#   * 2.14.0 only ships a manylinux_2_34_armv7l wheel, which needs glibc
#     >=2.34 (Debian Bookworm+). This Pi is on Bullseye (glibc 2.31), so
#     pip rejected it as incompatible even though the Python version and
#     architecture both matched -- confirmed by actually trying it.
#   * 2.13.0 (and 2.7.0-2.13.0 generally) ships a
#     manylinux2014_armv7l wheel instead, which only needs glibc >=2.17 --
#     comfortably covers Bullseye. This is the newest release that still
#     has that looser tag.
# Left unpinned, pip would default to 2.14.0 and fail the same way again
# (or worse, backtrack through releases the way plain `ultralytics` did,
# looking for one that resolves) -- pinning avoids both.
#
# If this Pi is ever upgraded to Bookworm or later, 2.14.0+ should work
# and can be tried again. Fallback if 2.13.0 somehow doesn't install:
# onnxruntime (see github.com/nknytk/built-onnxruntime-for-raspberrypi-linux
# for an armv7l-compatible build) instead of tflite-runtime.
tflite-runtime==2.13.0
